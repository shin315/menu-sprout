Vietnamese Lunar Calendar Widget Recipe

Capability

Display today's Vietnamese lunar calendar date (âm lịch): lunar day, month, and year, the year's sexagenary name (Can Chi), and whether the month is a leap month (tháng nhuận), for a menu-sprout widget.

Compute the lunar date locally using the astronomical algorithm below rather than scraping web pages. The calculation runs offline on device with no external network calls.

Scope: this recipe provides the full, self-contained astronomical conversion algorithm (UTC+7 Hanoi standard by Hồ Ngọc Đức based on Jean Meeus) and the year Can Chi formula. UI presentation uses standard Vietnamese calendar labels (Âm lịch, Dương lịch, Can Chi, Nhuận).

This recipe is self-contained. Do not require external source code, websites, or implementation guides to complete it.

Astronomical Algorithm and Formulas

The Vietnamese lunar calendar is an astronomical lunisolar calendar based on the movements of the Sun and the Moon, calculated for the UTC+7:00 time zone (Hanoi meridian 105°E). Below is the self-contained calculation logic to copy into vietnamese-lunar-calendar/lunar-calc.ts:

export function jdFromDate(dd: number, mm: number, yy: number): number {
  const a = Math.floor((14 - mm) / 12);
  const y = yy + 4800 - a;
  const m = mm + 12 * a - 3;
  let jd = dd + Math.floor((153 * m + 2) / 5) + 365 * y + Math.floor(y / 4) - Math.floor(y / 100) + Math.floor(y / 400) - 32045;
  if (jd < 2299161) {
    jd = dd + Math.floor((153 * m + 2) / 5) + 365 * y + Math.floor(y / 4) - 32083;
  }
  return jd;
}

function NewMoon(k: number): number {
  const T = k / 1236.85;
  const T2 = T * T;
  const T3 = T2 * T;
  const dr = Math.PI / 180;
  let Jd1 = 2415020.75933 + 29.53058868 * k + 0.0001178 * T2 - 0.000000155 * T3;
  Jd1 += 0.00033 * Math.sin((166.56 + 132.87 * T - 0.009173 * T2) * dr);
  const M = 359.2242 + 29.10535608 * k - 0.0000333 * T2 - 0.00000347 * T3;
  const Mpr = 306.0253 + 385.81691806 * k + 0.0107306 * T2 + 0.00001236 * T3;
  const F = 21.2964 + 390.67050646 * k - 0.0016528 * T2 - 0.00000239 * T3;
  let C1 = (0.1734 - 0.000393 * T) * Math.sin(M * dr) + 0.0021 * Math.sin(2 * dr * M);
  C1 = C1 - 0.4068 * Math.sin(Mpr * dr) + 0.0161 * Math.sin(dr * 2 * Mpr);
  C1 = C1 - 0.0004 * Math.sin(dr * 3 * Mpr);
  C1 = C1 + 0.0104 * Math.sin(dr * 2 * F) - 0.0051 * Math.sin(dr * (M + Mpr));
  C1 = C1 - 0.0074 * Math.sin(dr * (M - Mpr)) + 0.0004 * Math.sin(dr * (2 * F + M));
  C1 = C1 - 0.0004 * Math.sin(dr * (2 * F - M)) - 0.0006 * Math.sin(dr * (2 * F + Mpr));
  C1 = C1 + 0.0010 * Math.sin(dr * (2 * F - Mpr)) + 0.0005 * Math.sin(dr * (2 * Mpr + M));
  const deltat = T < -11
    ? 0.001 + 0.000839 * T + 0.0002261 * T2 - 0.00000845 * T3 - 0.000000081 * T * T3
    : -0.000278 + 0.000265 * T + 0.000262 * T2;
  return Jd1 + C1 - deltat;
}

function SunLongitude(jdn: number): number {
  const T = (jdn - 2451545.0) / 36525;
  const T2 = T * T;
  const dr = Math.PI / 180;
  const M = 357.52910 + 35999.05030 * T - 0.0001559 * T2 - 0.00000048 * T * T2;
  const L0 = 280.46645 + 36000.76983 * T + 0.0003032 * T2;
  let DL = (1.914600 - 0.004817 * T - 0.000014 * T2) * Math.sin(dr * M);
  DL += (0.019993 - 0.000101 * T) * Math.sin(dr * 2 * M) + 0.000290 * Math.sin(dr * 3 * M);
  let L = (L0 + DL) * dr;
  L = L - Math.PI * 2 * Math.floor(L / (Math.PI * 2));
  return L;
}

function getSunLongitude(dayNumber: number, timeZone = 7): number {
  return Math.floor(SunLongitude(dayNumber - 0.5 - timeZone / 24) / Math.PI * 6);
}

function getNewMoonDay(k: number, timeZone = 7): number {
  return Math.floor(NewMoon(k) + 0.5 + timeZone / 24);
}

function getLunarMonth11(yy: number, timeZone = 7): number {
  const off = jdFromDate(31, 12, yy) - 2415021;
  const k = Math.floor(off / 29.530588853);
  let nm = getNewMoonDay(k, timeZone);
  const sunLong = getSunLongitude(nm, timeZone);
  if (sunLong >= 9) nm = getNewMoonDay(k - 1, timeZone);
  return nm;
}

function getLeapMonthOffset(a11: number, timeZone = 7): number {
  const k = Math.floor((a11 - 2415021.076998695) / 29.530588853 + 0.5);
  let last = 0;
  let i = 1;
  let arc = getSunLongitude(getNewMoonDay(k + i, timeZone), timeZone);
  do {
    last = arc;
    i++;
    arc = getSunLongitude(getNewMoonDay(k + i, timeZone), timeZone);
  } while (arc !== last && i < 14);
  return i - 1;
}

export function convertSolar2Lunar(dd: number, mm: number, yy: number, timeZone = 7) {
  const dayNumber = jdFromDate(dd, mm, yy);
  const k = Math.floor((dayNumber - 2415021.076998695) / 29.530588853);
  let monthStart = getNewMoonDay(k + 1, timeZone);
  if (monthStart > dayNumber) monthStart = getNewMoonDay(k, timeZone);
  if (monthStart > dayNumber) monthStart = getNewMoonDay(k - 1, timeZone);
  let a11 = getLunarMonth11(yy, timeZone);
  let b11 = a11;
  let lunarYear: number;
  if (a11 >= monthStart) {
    lunarYear = yy;
    a11 = getLunarMonth11(yy - 1, timeZone);
  } else {
    lunarYear = yy + 1;
    b11 = getLunarMonth11(yy + 1, timeZone);
  }
  const lunarDay = dayNumber - monthStart + 1;
  const diff = Math.floor((monthStart - a11) / 29);
  let lunarLeap = 0;
  let lunarMonth = diff + 11;
  if (b11 - a11 > 365) {
    const leapMonthDiff = getLeapMonthOffset(a11, timeZone);
    if (diff >= leapMonthDiff) {
      lunarMonth = diff + 10;
      if (diff === leapMonthDiff) lunarLeap = 1;
    }
  }
  if (lunarMonth > 12) lunarMonth -= 12;
  if (lunarMonth >= 11 && diff < 4) lunarYear -= 1;
  return { lunarDay, lunarMonth, lunarYear, lunarLeap, jd: dayNumber };
}

Year Can Chi (Sexagenary Cycle)

Ten Heavenly Stems (Can): ["Giáp", "Ất", "Bính", "Đinh", "Mậu", "Kỷ", "Canh", "Tân", "Nhâm", "Quý"]

Twelve Earthly Branches (Chi): ["Tý", "Sửu", "Dần", "Mão", "Thìn", "Tỵ", "Ngọ", "Mùi", "Thân", "Dậu", "Tuất", "Hợi"]

For the lunar year returned by convertSolar2Lunar: Can index = (lunarYear + 6) % 10, Chi index = (lunarYear + 8) % 12. For example, lunar year 2026 is Bính Ngọ.

Implementation Contract

This is pure local computation with no filesystem, shell, network, credential, or token access, so it needs no server action. Copy the algorithm into vietnamese-lunar-calendar/lunar-calc.ts and call it from the widget.

The renderer widget lives in vietnamese-lunar-calendar/widget.tsx and vietnamese-lunar-calendar/components.tsx. It displays today's solar date and the lunar day, month, and year, the year Can Chi, and a Nhuận marker when the month is a leap month.

"Today" is the current calendar date in UTC+7 (Vietnam standard time), not the machine's local time zone.

Refresh: export a RefreshableMenuSproutWidget whose refreshView recomputes the snapshot so it is current each time the popover opens. The date changes once a day, so a slow visible interval such as 60 seconds is enough to catch midnight rollover while the popover is open. Do not add a background task or persist anything to db; the snapshot is cheap to recompute.

Before reporting work done, run the finished lunar-calc.ts against the known dates in the acceptance criteria and confirm the rendered widget shows the matching values.

Error Handling

  • The algorithm is valid for solar years 1800 to 2199. Outside that range, render an explicit unavailable state instead of a computed date.
  • Never render placeholder or mock lunar dates.

Security Constraints

  • Make no network requests; the calculation must run fully offline.
  • Do not read files, spawn commands, or use credentials.
